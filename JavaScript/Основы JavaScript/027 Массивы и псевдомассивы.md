
```
'use strict';

const arr = [1, 2, 3, 6, 8]

// Удаляет последний элемент массива
arr.pop(); 
console.log(`После удаления ${arr}`);

// Добавляет в конце массива новый элемент
arr.push(10); 
console.log(`После добавления ${arr}`);

// Перебор элементов массива по Index
for (let i=0; i < arr.length; i++) {
	console.log(`Перебор массива по индеку ${arr[i]}`);
}

// в for of - можно использовать break и continue
for (let value of arr) {
	console.log(`Перебор массива ${value}`);
}

// Определение количество элементов массиве. length работает по перебору от 0 до конца массива и в конце делает + 1 
console.log(`Определения количества элементов в массиве ${arr.length}`);

// arr.forEach позволяет перебрать элементы массива с возможностью передачи callback функцию. arr.forEach(function(элемент перебора, номер по порядку, массив).

arr.forEach(function(item, i, arr) {
	console.log(`${i}: ${item} внутри массива ${arr}`);
});

// str.split - позволяет раделить строку на отдельные элементы массива согласно разделителям.
const str = prompt("", "");
const products = str.split(", ");
console.log(products);

// arr.join - позволяет из массива сделать полноценную строку с разделителем.
const str = prompt("", "");
const products = str.split(", ");
console.log(products.join('; ));

// arr.sort - позволяет отсортировать элементы массива
const str = prompt("", "");
const products = str.split(", ");
products.sort();
console.log(products.join('; ));

// Сортировка чисел
const arr = [2, 13, 26, 8, 10];
arr.sort();
console.log(arr);

function compareNum(a, b) {
	return a - b;
}
```
